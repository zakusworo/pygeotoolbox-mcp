| Single-flash | One separator | multiflash.py | Double-flash = +30-50% power |

Usage Example:

from pygeotoolbox import multiflash, ncg, parasitic

# Double-flash cycle
cycle = multiflash.double_flash_cycle(
    m_total_kg_s=400,
    T_separator_C=180,
    P_separator_kPa=1000,
    T_flash2_C=140,
    P_flash2_kPa=361,
)

# NCG penalty
ncg_penalty = ncg.total_ncg_penalty(m_steam_kg_s=cycle['m_steam_total_kg_s'])

# Parasitic loads
net = parasitic.parasitic_loads(
    gross_power_MW=cycle['net_MW'],
    m_steam_kg_s=cycle['m_steam_total_kg_s'],
    m_brine_kg_s=400 - cycle['m_steam_total_kg_s'],
)

print(f"Net: {net['net_power_MW']:.1f} MW")
